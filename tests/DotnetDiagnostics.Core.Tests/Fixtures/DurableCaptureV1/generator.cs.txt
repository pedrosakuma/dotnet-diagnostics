using System.IO.Compression;
using System.Security.Cryptography;
using System.Text.Json;
using DotnetDiagnostics.Core.Artifacts;
using DotnetDiagnostics.Core.Captures;

var root = Path.GetFullPath(args[0]);
var fixtures = Path.GetFullPath(args[1]);
var generatorSource = Path.GetFullPath(args[2]);
Directory.CreateDirectory(fixtures);
var store = new SqliteCaptureStore(new FixtureRoot(root));
var access = new CaptureAccess("fixture-owner-v1");
var timestamp = new DateTimeOffset(2026, 9, 24, 0, 0, 0, TimeSpan.Zero);

await using var sealedWriter = await store.CreateAsync(new("Frozen sealed v1 fixture", "compatibility"), access);
var sealedArtifact = sealedWriter.AddArtifact("synthetic", "Synthetic v1 occurrences");
if (!sealedWriter.TryAppend(sealedArtifact, new(timestamp, 7, "fixture", "synthetic-event", 1.25, 123, "count",
    [new("message", CaptureFieldKind.Text, StringValue: "frozen-v1 🧪"),
     new("integer", CaptureFieldKind.SignedInteger, Int64Value: long.MaxValue)])) ||
    !sealedWriter.TryAppend(sealedArtifact, new(Name: "missing-dimensions",
        Fields: [new("explicit-null", CaptureFieldKind.Null)])))
    throw new InvalidOperationException("Synthetic fixture admission failed.");
sealedWriter.SetSnapshot(sealedArtifact, 1, "{\"fixtureVersion\":1,\"synthetic\":true}"u8.ToArray());
sealedWriter.SetSourceRejected(0);
var sealedInfo = await sealedWriter.CompleteAsync();
using (var reader = await store.OpenAsync(sealedInfo.CaptureId, access))
{
    if (reader.Query(new(sealedArtifact)).Records.Count != 2 || !reader.Info.Quality.IsComplete ||
        reader.ReadSnapshot(sealedArtifact)?.Version != 1)
        throw new InvalidOperationException("The committed v1 reader did not verify its generated sealed fixture.");
}

var interruptedWriter = await store.CreateAsync(new("Frozen interrupted v1 fixture", "compatibility"), access);
var interruptedArtifact = interruptedWriter.AddArtifact("synthetic", "Interrupted synthetic v1 occurrences");
if (!interruptedWriter.TryAppend(interruptedArtifact,
    new(timestamp, 8, "fixture", "committed-before-disposal", 2)))
    throw new InvalidOperationException("Interrupted fixture admission failed.");
await interruptedWriter.DisposeAsync();
var interruptedInfo = (await store.ListAsync(access)).Captures.Single(c => c.CaptureId == interruptedWriter.Reference.CaptureId);

var packages = new[]
{
    Freeze("sealed-v1.zip", sealedInfo, sealedArtifact),
    Freeze("interrupted-v1.zip", interruptedInfo, interruptedArtifact)
};
File.Copy(generatorSource, Path.Combine(fixtures, "generator.cs.txt"), overwrite: false);
var provenance = new
{
    ProducerCommit = "59e34e40681720334db3812f5746e463592449d0",
    ProducerSdk = "10.0.201",
    ProducerCoreAssemblySha256 = Hash(typeof(SqliteCaptureStore).Assembly.Location),
    GeneratorSourceSha256 = Hash(generatorSource),
    GeneratedUtc = DateTimeOffset.UtcNow,
    Origin = "Synthetic scalar fixtures generated by the committed v1 public store API before any v2 implementation.",
    Verification = "The committed v1 reader opened/query-verified the sealed fixture. No previous-reader behavior is inferred or fabricated.",
    PackageVersions = new { Package = 1, Schema = 1, Record = 1, Index = 1, Writer = 1, Reader = 1 },
    RequiredFeatures = new[] { "normalized-scalars-v1" },
    StoreMarker = "dotnet-diagnostics-captures/1",
    SnapshotRepresentationVersion = 1,
    Packages = packages
};
await File.WriteAllTextAsync(Path.Combine(fixtures, "provenance.json"),
    JsonSerializer.Serialize(provenance, new JsonSerializerOptions { WriteIndented = true }));
Console.WriteLine(JsonSerializer.Serialize(provenance));

object Freeze(string archiveName, CaptureInfo info, string artifactId)
{
    var directory = Path.Combine(root, "captures", info.CaptureId);
    var archive = Path.Combine(fixtures, archiveName);
    ZipFile.CreateFromDirectory(directory, archive, CompressionLevel.SmallestSize, includeBaseDirectory: false);
    return new
    {
        Archive = archiveName,
        ArchiveSha256 = Hash(archive),
        info.CaptureId,
        info.OwnerId,
        ArtifactId = artifactId,
        State = info.State.ToString(),
        Members = Directory.EnumerateFiles(directory).Order(StringComparer.Ordinal).ToDictionary(
            static path => Path.GetFileName(path), static path => Hash(path), StringComparer.Ordinal)
    };
}

static string Hash(string path)
{
    using var input = File.OpenRead(path);
    return Convert.ToHexString(SHA256.HashData(input));
}

internal sealed record FixtureRoot(string Root) : IArtifactRootProvider;
